const express = require('express'); const cors = require('cors'); const jwt = require('jsonwebtoken'); const app = express(); const PORT = process.env.PORT || 3000; const JWT_SECRET = 'adpulse_secret_key_2026_secured'; app.use(cors()); app.use(express.json()); // ========================================== // IN-MEMORY DATABASES (AdPulse Network) // ========================================== let users = []; // यूज़र्स (Advertisers & Publishers) let platforms = []; // रजिस्टर्ड वेबसाइट्स / ऐप्स / यूट्यूब चैनल्स let ads = []; // विज्ञापन अभियान (Campaigns) let clickLogs = {}; // एंटी-फ्रॉड क्लिक ट्रैकिंग (IP Based) // Minimum Payout Threshold (₹1000 Rule) const MIN_WITHDRAWAL_LIMIT = 1000; // ========================================== // 1. CUSTOM AUTHENTICATION (Name & Password Rules) // ========================================== // (A) Signup / Registration app.post('/api/auth/register', (req, res) => { const { fullName, email, password, role } = req.body; if (!fullName || !email || !password) { return res.status(400).json({ success: false, message: "सभी फ़ील्ड (पूरा नाम, ईमेल, पासवर्ड) भरना अनिवार्य है।" }); } // पासवर्ड नियम: 8 से 10 अक्षर अनिवार्य if (password.length < 8 || password.length > 10) { return res.status(400).json({ success: false, message: "सुरक्षा नियम: पासवर्ड कम से कम 8 अक्षर और ज्यादा से ज्यादा 10 अक्षर का ही होना चाहिए।" }); } const existingUser = users.find(u => u.email === email); if (existingUser) { return res.status(400).json({ success: false, message: "यह ईमेल पहले से रजिस्टर्ड है।" }); } const newUser = { id: "usr_" + Date.now(), fullName, email, password, role: role || 'publisher', // 'publisher' या 'advertiser' balance: 0.00, // पब्लिशर की कुल कमाई createdAt: new Date() }; users.push(newUser); res.json({ success: true, message: "AdPulse Network में आपका अकाउंट सफलतापूर्वक बन गया है!", userId: newUser.id }); }); // (B) Login Verification app.post('/api/auth/login', (req, res) => { const { email, password } = req.body; const user = users.find(u => u.email === email && u.password === password); if (!user) { return res.status(401).json({ success: false, message: "गलत ईमेल या पासवर्ड।" }); } const token = jwt.sign({ id: user.id, email: user.email, role: user.role }, JWT_SECRET, { expiresIn: '7d' }); res.json({ success: true, message: "लॉगिन सफल रहा!", token, user: { id: user.id, fullName: user.fullName, email: user.email, role: user.role, balance: user.balance } }); }); // ========================================== // 2. PLATFORM ONBOARDING & VERIFICATION LOGIC // ========================================== app.post('/api/platform/add', (req, res) => { const { userId, platformUrl, platformType, subscriberCount } = req.body; // platformType: 'youtube', 'single_website', 'app' if (!platformUrl || !platformType) { return res.status(400).json({ success: false, message: "प्लेटफ़ॉर्म URL और टाइप अनिवार्य है।" }); } let status = "APPROVED"; let message = "बधाई हो! आपकी वेबसाइट तुरंत अप्रूव हो गई है। आप अभी एड लगा सकते हैं।"; // YouTube / बड़े प्लेटफ़ॉर्म्स के लिए एलिजिबिलिटी चेक if (platformType === 'youtube') { const subs = parseInt(subscriberCount) || 0; if (subs < 1000) { status = "PENDING"; message = "YouTube के लिए कम से कम 1000 सब्सक्राइबर्स अनिवार्य हैं। एलिजिबिलिटी पूरी होने के बाद अप्रूवल मिलेगा।"; } else { status = "APPROVED"; message = "YouTube चैनल एलिजिबिलिटी के अनुसार अप्रूव कर दिया गया है!"; } } // सिंगल/इंडिपेंडेंट वेबसाइट्स के लिए इंस्टेंट अप्रूवल (Zero Traffic Allowed) else if (platformType === 'single_website') { status = "APPROVED"; message = "आपकी वेबसाइट को AdPulse Network पर तुरंत स्वीकृति (Instant Approval) दे दी गई है!"; } const newPlatform = { platformId: "plt_" + Date.now(), userId, platformUrl, platformType, status, createdAt: new Date() }; platforms.push(newPlatform); res.json({ success: true, message, platform: newPlatform, adTag: status === "APPROVED" ? `` : null }); }); // ========================================== // 3. ADVERTISER CAMPAIGN MANAGEMENT // ========================================== app.post('/api/ads/create', (req, res) => { const { advertiserId, title, bannerUrl, targetUrl, size, cpc } = req.body; // size: '300x250', '728x90', '160x600' if (!title || !bannerUrl || !targetUrl) { return res.status(400).json({ success: false, message: "विज्ञापन की पूरी जानकारी भरें।" }); } const newAd = { adId: "ad_" + Date.now(), advertiserId, title, bannerUrl, targetUrl, size: size || '300x250', cpc: parseFloat(cpc) || 2.00, // Cost Per Click impressions: 0, clicks: 0 }; ads.push(newAd); res.json({ success: true, message: "विज्ञापन अभियान सफलतापूर्वक चालू हो गया!", ad: newAd }); }); // ========================================== // 4. AD SERVING & ANTI-FRAUD CLICK ENGINE // ========================================== // सर्व एड (Serve Ad) app.get('/api/serve-ad', (req, res) => { const { size } = req.query; const reqSize = size || '300x250'; const matchingAds = ads.filter(a => a.size === reqSize); const selectedAd = matchingAds.length > 0 ? matchingAds[0] : ads[0]; if (!selectedAd) { return res.status(404).send("AdPulse: No Active Ads Available"); } selectedAd.impressions += 1; const adHtml = `

${selectedAd.title}

 Ads by AdPulse Network
`; res.send(adHtml); }); // क्लिक ट्रैकिंग और फ्रॉड प्रोटेक्शन (Anti-Fraud Click Protection) app.get('/api/track-click', (req, res) => { const { adId, publisherId } = req.query; const userIp = req.ip || req.connection.remoteAddress; const ad = ads.find(a => a.adId === adId); if (!ad) return res.status(404).send("Ad Not Found"); const now = Date.now(); const clickKey = `${userIp}_${adId}`; // यदि एक ही IP से 1 मिनट के भीतर दोबारा क्लिक हुआ तो ब्लॉक करें (Anti-Fraud) if (clickLogs[clickKey] && (now - clickLogs[clickKey] < 60000)) { console.log(`[AdPulse Anti-Fraud] Blocked Invalid Click from IP: ${userIp}`); return res.redirect(ad.targetUrl); // यूज़र को रीडायरेक्ट कर दें लेकिन पैसे काउंट न करें } // वैलिड क्लिक दर्ज करें clickLogs[clickKey] = now; ad.clicks += 1; // पब्लिशर के खाते में पैसे जोड़ें (जैसे ₹2 CPC का 60% यानी ₹1.20 पब्लिशर को) if (publisherId) { const publisher = users.find(u => u.id === publisherId); if (publisher) { publisher.balance += (ad.cpc * 0.60); } } res.redirect(ad.targetUrl); }); // ========================================== // 5. PAYOUT & WITHDRAWAL SYSTEM (₹1000 Limit) // ========================================== app.post('/api/payout/request', (req, res) => { const { userId, upiIdOrBank } = req.body; const user = users.find(u => u.id === userId); if (!user) { return res.status(404).json({ success: false, message: "यूज़र नहीं मिला।" }); } // ₹1000 विड्रॉल रूल if (user.balance < MIN_WITHDRAWAL_LIMIT) { return res.status(400).json({ success: false, message: `न्यूनतम विड्रॉल सीमा ₹${MIN_WITHDRAWAL_LIMIT} है। आपकी वर्तमान कमाई: ₹${user.balance.toFixed(2)}` }); } const requestedAmount = user.balance; user.balance = 0.00; // बैलेंस रीसेट res.json({ success: true, message: `₹${requestedAmount.toFixed(2)} के विड्रॉल का अनुरोध स्वीकार कर लिया गया है। राशि आपके बैंक/UPI (${upiIdOrBank}) में 24 घंटे में भेज दी जाएगी।` }); }); // ========================================== // SERVER START // ========================================== app.listen(PORT, () => { console.log(`AdPulse Network Engine running successfully on http://localhost:${PORT}`); });
